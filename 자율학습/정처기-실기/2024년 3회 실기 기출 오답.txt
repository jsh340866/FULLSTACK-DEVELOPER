
#include <stdio.h>
void func(int **arr, int size){
    int i;
    for(i=0;i<size;i++){
        *(*arr+i)=(*(*arr+i)+i) % size;
    }
}
int main(){
    int arr[]={3, 1, 4, 1, 5};
    int *p=arr;
    int **pp=&p;

    func(pp, 5);
    printf("%d", arr[2]);
    return 0;
}
