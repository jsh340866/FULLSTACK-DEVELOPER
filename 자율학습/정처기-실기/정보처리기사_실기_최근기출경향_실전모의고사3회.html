#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int v;
    struct Node *next;
} Node;

Node* append(Node *head, int v) {
    Node *n = (Node*)malloc(sizeof(Node));
    n->v = v;
    n->next = NULL;

    if (head == NULL)
        return n;

    Node *p = head;
    while (p->next != NULL)
        p = p->next;

    p->next = n;
    return head;
}

Node* move_front(Node *head, int k) {
    if (head == NULL || head->v == k)
        return head;

    Node *p = head;

    while (p->next != NULL && p->next->v != k)
        p = p->next;

    if (p->next != NULL) {
        Node *target = p->next;
        p->next = target->next;
        target->next = head;
        head = target;
    }

    return head;
}

int main() {
    Node *head = NULL;

    for (int i = 1; i <= 5; i++)
        head = append(head, i);

    head = move_front(head, 4);

    for (Node *p = head; p != NULL; p = p->next)
        printf("%d", p->v);

    return 0;
}

        n->c = s[i];
        n->next = head;
        head = n;
    }

    while (head != NULL) {
        putchar(head->c);
        Node *t = head;
        head = head->next;
        free(t);
    }

    return 0;
}

class Node:
    def __init__(self, v):
        self.v = v
        self.children = []

def tree(values):
    nodes = [Node(v) for v in values]

    for i in range(1, len(values)):
        parent = (i - 1) // 2
        nodes[parent].children.append(nodes[i])

    return nodes[0]

def calc(node, level=0):
    value = node.v if level % 2 == 0 else 0

    return value + sum(
        calc(child, level + 1)
        for child in node.children
    )

root = tree([2, 4, 6, 8, 10, 12, 14])
print(calc(root))

typedef struct Node {
    int v;
    struct Node *next;
} Node;

Node* append(Node *head, int v) {
    Node *n = (Node*)malloc(sizeof(Node));
    n->v = v;
    n->next = NULL;

    if (head == NULL)
        return n;

    Node *p = head;
    while (p->next != NULL)
        p = p->next;

    p->next = n;
    return head;
}

Node* reverse_n(Node *head, int n) {
    Node *prev = NULL;
    Node *cur = head;
    Node *next = NULL;
    int count = 0;

    while (cur != NULL && count < n) {
        next = cur->next;
        cur->next = prev;
        prev = cur;
        cur = next;
        count++;
    }

    head->next = cur;
    return prev;
}

int main() {

    for (int i = 1; i <= 5; i++)
        head = append(head, i);

    head = reverse_n(head, 3);

    for (Node *p = head; p != NULL; p = p->next)
        printf("%d", p->v);

    return 0;

10번 정답 문제로 돌아가기
① 제2정규형(2NF)
② B → C에 의한 이행적 종속
③ R1(A,B,D), R2(B,C)
기본키가 단일 속성이므로 부분 함수 종속은 없다. 그러나 A→B이고 B→C이므로 C가 A에 이행적으로 종속되어 3NF를 위반한다.

13번 정답 문제로 돌아가기
① 정규 응답 모드(NRM) ② 비동기 응답 모드(ARM) ③ 비동기 균형 모드(ABM)
NRM과 ARM은 주국·종국의 불균형 구성이고, ABM은 두 복합국이 대등한 균형 구성이다.

① Strategy ② Factory Method
